public function sanitize_params() {
    $attributes = $this->get_attributes();
    if ( empty( $attributes['args'] ) ) {
        return true;                        // 没绑 schema → 一个字不洗
    }
    $order = $this->get_parameter_order();  // JSON → POST → GET → URL → defaults

    foreach ( $order as $type ) {
        foreach ( $this->params[ $type ] as $key => $value ) {
            if ( ! isset( $attributes['args'][ $key ] ) ) {
                continue;                   // ★ 未注册的参数原样保留，既不校验也不清洗
            }
            $param_args = $attributes['args'][ $key ];

            // 有 type 但没有 sanitize_callback → 默认注入 rest_parse_request_arg
            if ( ! array_key_exists( 'sanitize_callback', $param_args ) && ! empty( $param_args['type'] ) ) {
                $param_args['sanitize_callback'] = 'rest_parse_request_arg';
            }
            if ( empty( $param_args['sanitize_callback'] ) ) {
                continue;                   // 没有 type 也没有 callback → 跳过
            }

            $sanitized_value = call_user_func( $param_args['sanitize_callback'], $value, $this, $key );
            if ( is_wp_error( $sanitized_value ) ) {
                $invalid_params[ $key ] = ...;          // 不抛出，继续洗别的
            } else {
                $this->params[ $type ][ $key ] = $sanitized_value;   // ★ 就地改写
            }
        }
    }
    if ( $invalid_params ) {
        return new WP_Error( 'rest_invalid_param', ..., array( 'status' => 400, 'params' => ..., 'details' => ... ) );
    }
    return true;
}

function rest_validate_request_arg( $value, $request, $param ) {
	$attributes = $request->get_attributes();
	if ( ! isset( $attributes['args'][ $param ] ) || ! is_array( $attributes['args'][ $param ] ) ) {
		return true;

	$args = $attributes['args'][ $param ];

}

protected function match_request_to_handler( $request ) {
		$method = $request->get_method();
		$path   = $request->get_route();

		$with_namespace = array();

		foreach ( $this->get_namespaces() as $namespace ) {
			if ( str_starts_with( trailingslashit( ltrim( $path, '/' ) ), $namespace ) ) {
				$with_namespace[] = $this->get_routes( $namespace );
			}
		}

		if ( $with_namespace ) {
			$routes = array_merge( ...$with_namespace );
		} else {
			$routes = $this->get_routes();
		}

		foreach ( $routes as $route => $handlers ) {
			$match = preg_match( '@^' . $route . '$@i', $path, $matches );

			if ( ! $match ) {
				continue;
			}

			$args = array();

			foreach ( $matches as $param => $value ) {
				if ( ! is_int( $param ) ) {
					$args[ $param ] = $value;
				}
			}

			foreach ( $handlers as $handler ) {
				$callback = $handler['callback'];

				// Fallback to GET method if no HEAD method is registered.
				$checked_method = $method;
				if ( 'HEAD' === $method && empty( $handler['methods']['HEAD'] ) ) {
					$checked_method = 'GET';
				}
				if ( empty( $handler['methods'][ $checked_method ] ) ) {
					continue;
				}

				if ( ! is_callable( $callback ) ) {
					return array( $route, $handler );
				}

				$request->set_url_params( $args );
				$request->set_attributes( $handler );

				$defaults = array();

				foreach ( $handler['args'] as $arg => $options ) {
					if ( isset( $options['default'] ) ) {
						$defaults[ $arg ] = $options['default'];
					}
				}

				$request->set_default_params( $defaults );

				return array( $route, $handler );
			}
		}

		return new WP_Error(...);
	}

大概就是把全局已经注册的路由和这里对应上呢

这里callback GET调用的就是get_items，此时就会触发Wp_Query对象的实例化以及对query函数的调用

我们仔细看看get_items的实现，会发现一个很恐怖的问题

callback的函数对处理的request是完全信任的，除了只检查 orderby/include 前置条件和 context=edit 权限

对于其他的参数几乎不做校验，接近于直接处理body了

但是这样可以跑通了吗?

好像不行呢。。我们发现/v1/batch路由对请求体的request method做了严格的校验

这里的校验还是在sanitize_params中

在这里有检测enum的分支

直接无法通过

那么我们就要想办法绕过它去做GET的请求

这里感觉也是作者和他的AI最天才的地方啊，使用一个嵌套去做绕过

此时顶层 POST /batch/v1 经 dispatch 的 batch schema 校验合法进入回调 serve_batch_request_v1

outer[0] 的 URL 解析失败使 $matches 错位一格，outer[1] 借到 outer[2] 匹配的 /batch/v1 handler，其 body 里前面作为未知参数被sanitize放行的内层 batch 被当作批处理输入再跑一轮——这层绕过的是 method 枚举（枚举只存在于顶层路由的 schema，内层信封结构无 schema）

内层子请求各自在循环二绑上自己匹配到的 schema，inner[1]（DELETE /wp/v2/posts/1）由于绑定的 delete_item schema 不认识 author_exclude在校验中直接进行了保留

内层同样的错位让 inner[1] 在执行时借到 inner[2] 的 get_items 匹配 → 这层绕过的是参数校验（配对不一致）； get_items 读走DELETE body里未清洗的标量 author_exclude → 映射 author__not_in → 实例化 WP_Query → query() → get_posts() 的“仅数组消毒”分支原样拼接 → $wpdb->get_results() 注入达成。
